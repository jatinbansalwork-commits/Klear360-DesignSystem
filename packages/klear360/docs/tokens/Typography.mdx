> These tokens should be used along with the [makeTypographySize util](?path=/story/utils-maketypographysize--docs).

export const Typography = () => {
  const { theme } = useTheme();
  return (
    <>
      <h3>Font Family</h3>
      <table>
        <thead>
          <tr>
            <th>Token Name</th>
            <th>Value</th>
          </tr>
        </thead>
        <tbody>
          {Object.entries(theme.typography.fonts.family).map(([key, value]) => (
            <tr key={key}>

              <td>{value}</td>
            </tr>
          ))}
        </tbody>
      </table>
      <h3>Font Weight</h3>
      <table>
        <thead>
          <tr>
            <th>Token Name</th>
            <th>Value</th>
          </tr>
        </thead>
        <tbody>
          {Object.entries(theme.typography.fonts.weight).map(([key, value]) => (
            <tr key={key}>
              <td>{`theme.typography.fonts.weight.${key}`}</td>
              <td>{value}</td>
            </tr>
          ))}
        </tbody>
      </table>
      <h3>Font Size</h3>
      <table>
        <thead>
          <tr>
            <th>Token Name</th>
            <th />
          </tr>
        </thead>
        <tbody>
          {Object.entries(theme.typography.fonts.size).map(([key, value]) => (

    </>

};
